R: leaq pega um endereço (a partir de um deslocamento de registrador) e coloca
   essa memória em um registrador; ele carrega o endereço, não o valor.
   É usado geralmente com ponteiross


/*
int novonum(void) {
  int minhalocal;
  printf("numero: ");
  scanf("%d",&minhalocal);
  return minhalocal;
}
*/
Sf1: .string "%d"
Sf2: .string "Numero: "
.text
.globl novonum
novonum:
    pushq %rbp
    movq %rsp, %rbp
    subq $16, %rsp

    movq %rdi, -8(%rbp)
    movq $Sf2, %rdi
    call printf
    movq -8(%rbp), %rdi

    movq $Sf1, %rdi
    leaq -4(%rbp), %rsi
    call scanf
    movl -4(%rbp), %eax

    leave
    ret


/*
int bemboba (int num) {
  int local[4];
  int *a;
  int i;

  for (i=0,a=local;i<4;i++) {
    *a = num;
    a++;
  }
  return addl (local, 4);
}
*/

.text 
.globl bemboba
bemboba:
    pushq %rbp
    movq %rsp, %rbp
    subq $32, %rsp

    movq %rbx, -8(%rbp)     # rbx = i
    movq %r12, -16(%rbp)    #r12 = a
    
    movl $0, %ebx    # i=0 
    
    leaq -32(%rbp), %r12    # endereço de local[0] (array começa em -32(%rbp))

L1:
    cmpl $4, %ebx 
    jge L2

    movl %edi, (%r12)

    addq $4, %r12

    incl %ebx

    jmp L1

L2:
    leaq -32(%rbp), %rdi
    movl $4, %esi
    call addl

    movq -8(%rbp), %rbx
    movq -16(%rbp), %rbx

    leave
    ret
